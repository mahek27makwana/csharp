# 3. Create Cart Controller

Create:

```text
Controllers/CartController.cs
```

### `CartController.cs`

```csharp
using Microsoft.AspNetCore.Mvc;
using ShoppingCart.Models;

namespace ShoppingCart.Controllers
{
    public class CartController : Controller
    {
        static List<Product> products = new List<Product>
        {
            new Product { Id = 1, Name = "Laptop", Price = 50000 },
            new Product { Id = 2, Name = "Mobile", Price = 20000 },
            new Product { Id = 3, Name = "Headphones", Price = 2000 }
        };

        static List<Product> cart = new List<Product>();

        public IActionResult Index()
        {
            return View(products);
        }

        public IActionResult Add(int id)
        {
            var product = products.FirstOrDefault(x => x.Id == id);

            if (product != null)
                cart.Add(product);

            return RedirectToAction("Cart");
        }

        public IActionResult Cart()
        {
            return View(cart);
        }

        public IActionResult Remove(int id)
        {
            var product = cart.FirstOrDefault(x => x.Id == id);

            if (product != null)
                cart.Remove(product);

            return RedirectToAction("Cart");
        }
    }
}

# 4. Create Product View

Create folder:

```text
Views/Cart
```

Create:

```text
Views/Cart/Index.cshtml
```

### `Index.cshtml`

```html
@model List<ShoppingCart.Models.Product>

<h2>Products</h2>

@foreach (var p in Model)
{
    <p>
        <b>@p.Name</b> - ₹@p.Price
        <a asp-action="Add" asp-route-id="@p.Id">Add to Cart</a>
    </p>
}

<a asp-action="Cart">View Cart</a>
# 5. Create Cart View

Create:

```text
Views/Cart/Cart.cshtml
```

### `Cart.cshtml`

```html
@model List<ShoppingCart.Models.Product>

<h2>Shopping Cart</h2>

@foreach (var p in Model)
{
    <p>
        @p.Name - ₹@p.Price
        <a asp-action="Remove" asp-route-id="@p.Id">Remove</a>
    </p>
}

<h3>Total: ₹@Model.Sum(x => x.Price)</h3>

<a asp-action="Index">Continue Shopping</a>

# 6. Change `Program.cs`

Open `Program.cs` and use:

```csharp
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllersWithViews();

var app = builder.Build();

app.UseStaticFiles();
app.UseRouting();

app.MapControllerRoute(
    name: "default",
    pattern: "{controller=Cart}/{action=Index}/{id?}");

app.Run();
